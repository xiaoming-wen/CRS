        const newHeight = this.resizeStartHeight + (e.clientY - this.resizeStartY)

    generateGraphicElements(dom, data) {
      const chartWidth = dom.clientWidth
      const chartHeight = dom.clientHeight
      const centerX = chartWidth / 2
      const centerY = chartHeight / 2
      const minDim = Math.min(chartWidth, chartHeight)
      const innerRadius = minDim * 0.17 / 2  // 最低分对应半径
      const outerRadius = minDim * 0.4 / 2   // 最高分对应半径
      const arrowStartRadius = innerRadius + 2

      // 计算总分
      const total = this.showScores
        ? data.reduce((sum, item) => sum + item.value, 0)
        : data.reduce((sum, item) => sum + item.value, 0)

      if (total === 0) return []

      let startAngle = -Math.PI / 2 - Math.PI / 3
      const graphics = []
      const targetColors = {
        '关键能力': '#5B9BD5',
        '关键行为': '#ED7D31',
        '情感体验': '#A4A4A4'
      }

      // 获取各项满分值
      const maxScores = {
        '关键能力': 100,
        '关键行为': 100,
        '情感体验': 100
      }
      data.forEach((item, index) => {
        let midAngle
        if (this.showScores) {
          // 显示得分时使用固定角度
          const angles = [
            -Math.PI / 2,                    // 顶部顶点角度
            -Math.PI / 2 + 2 * Math.PI / 3,  // 右下顶点角度
            -Math.PI / 2 + 4 * Math.PI / 3   // 左下顶点角度
          ]
          midAngle = angles[index % angles.length]
        } else {
          // 不显示得分时计算中间角度
          const ratio = 1 / 3
          const endAngle = startAngle + ratio * Math.PI * 2
          midAngle = (startAngle + endAngle) / 2
          startAngle = endAngle
        }

        // 计算标签位置（控制在圆圈范围内）
        let labelRadius
        if (item.name === '情感体验') {
          labelRadius = outerRadius * 1.25 // 增大半径，远离内圈
        } else {
          labelRadius = outerRadius * 1.4 // 保持原有距离
        }
        const labelX = centerX + Math.cos(midAngle) * labelRadius
        const labelY = centerY + Math.sin(midAngle) * labelRadius

        let rotationDeg = (midAngle * 180 / Math.PI) + 90

        if (rotationDeg > 90 && rotationDeg < 270) {
          rotationDeg += 180 // 翻转180度避免文字倒置
        }
        if (item.name === '关键行为') {
          rotationDeg = 120 // 左下标签微调
        } else if (item.name === '情感体验') {
          rotationDeg = 60 // 右下标签微调
        } else if (item.name === '关键能力') {
          rotationDeg = 0 // 顶部标签保持水平
        }

        // 添加旋转的文字标签
        graphics.push({
          type: 'text',
          style: {
            text: item.name,
            fill: '#FFFFFF',
            fontSize: 16,
            fontWeight: 'bold',
            align: 'center',
            baseline: 'middle'
          },
          position: [labelX, labelY],
          rotation: rotationDeg * Math.PI / 180, // 转为弧度
          z: 13 // 确保在最上层显示
        })
      })

      // 显示得分时 - 动态计算顶点位置和连接线
      if (this.showScores) {
        const points = []
        const angles = [
          -Math.PI / 2,                    // 顶部顶点角度
          -Math.PI / 2 + 2 * Math.PI / 3,  // 右下顶点角度
          -Math.PI / 2 + 4 * Math.PI / 3   // 左下顶点角度
        ]

        data.forEach((item, index) => {
          const maxScore = maxScores[item.name] || 100
          const scoreRatio = Math.min(Math.max(item.value / maxScore, 0), 1)
          const currentRadius = innerRadius + (outerRadius - innerRadius) * scoreRatio
          const midAngle = angles[index % angles.length]

          const endX = centerX + Math.cos(midAngle) * currentRadius
          const endY = centerY + Math.sin(midAngle) * currentRadius
          const lineColor = targetColors[item.name] || '#ffffff'

          // 绘制从内圆到顶点的连接线
          graphics.push({
            type: 'line',
            shape: {
              x1: centerX + Math.cos(midAngle) * arrowStartRadius,
              y1: centerY + Math.sin(midAngle) * arrowStartRadius,
              x2: endX,
              y2: endY
            },
            style: {
              stroke: '#FFFFFF',
              lineWidth: 2,
              lineDash: [4, 4]
            },
            z: 10
          })

          // 绘制折线延伸（用于显示得分）
          const extensionLength = outerRadius * 1.2
          const extendedX = centerX + Math.cos(midAngle) * (currentRadius + extensionLength)
          const extendedY = centerY + Math.sin(midAngle) * (currentRadius + extensionLength)

          graphics.push({
            type: 'polyline',
            shape: {
              points: [
                [endX, endY],
                [extendedX, extendedY]
              ]
            },
            style: {
              stroke: '#ffffff',
              lineWidth: 0.5,
              lineDash: [2, 2]
            },
            z: 11
          })

          // 显示得分文本
          graphics.push({
            type: 'text',
            style: {
              text: Math.round(item.value),
              fill: lineColor,
              fontSize: 14,
              fontWeight: 'bold',
              align: 'center',
              verticalAlign: 'middle'
            },
            position: [extendedX, extendedY],
            z: 12
          })

          points.push({ x: endX, y: endY })
        })

        // 绘制三角形的三条边
        if (data.length === 3) {
          for (let i = 0; i < points.length; i++) {
            const currentPoint = points[i]
            const nextPoint = points[(i + 1) % points.length]

            graphics.push({
              type: 'line',
              shape: {
                x1: currentPoint.x,
                y1: currentPoint.y,
                x2: nextPoint.x,
                y2: nextPoint.y
              },
              style: {
                stroke: 'rgba(122, 202, 238, 0.6)',
                lineWidth: 2,
              },
              z: 9
            })
          }
        }
      } else {
        // 不显示得分时 - 固定比例逻辑
        data.forEach(item => {
          const ratio = 1 / 3
          const endAngle = startAngle + ratio * Math.PI * 2
          const midAngle = (startAngle + endAngle) / 2

          const startX = centerX + Math.cos(midAngle) * arrowStartRadius
          const startY = centerY + Math.sin(midAngle) * arrowStartRadius
          const endX = centerX + Math.cos(midAngle) * outerRadius
          const endY = centerY + Math.sin(midAngle) * outerRadius

          const lineColor = targetColors[item.name] || '#7ACAEE'

          // 绘制直线
          graphics.push({
            type: 'line',
            shape: {
              x1: startX,
              y1: startY,
              x2: endX,
              y2: endY
            },
            style: {
              stroke: lineColor,
              lineWidth: 5
            },
            z: 10
          })

          // 绘制箭头
          const arrowSize = 8
          const arrowAngle = Math.atan2(endY - startY, endX - startX)

          const arrowPoint1 = [endX, endY]
          const arrowPoint2 = [
            endX - arrowSize * Math.cos(arrowAngle - Math.PI / 6),
            endY - arrowSize * Math.sin(arrowAngle - Math.PI / 6)
          ]
          const arrowPoint3 = [
            endX - arrowSize * Math.cos(arrowAngle + Math.PI / 6),
            endY - arrowSize * Math.sin(arrowAngle + Math.PI / 6)
          ]

          graphics.push({
            type: 'polygon',
            shape: {
              points: [arrowPoint1, arrowPoint2, arrowPoint3]
            },
            style: {
              fill: lineColor,
              stroke: lineColor,
              lineWidth: 1
            },
            z: 11
          })

          startAngle = endAngle
        })
      }

      return graphics
    },

    // 格式化综合评价文本
    formatReviewText(text) {
      if (!text) return ''

      // 1. 按段落分割
      const paragraphs = text.split('\n').filter(p => p.trim())

      // 2. 对每个段落进行处理
      const formattedParagraphs = paragraphs.map(paragraph => {
        // 移除首尾空格
        let formatted = paragraph.trim()

        // 在段落头部添加两个全角空格（中文缩进）
        formatted = `　　${formatted}`

        // 优化标点符号（确保中文标点）
        formatted = formatted
          .replace(/,/g, '，')
          .replace(/\./g, '。')
          .replace(/\?/g, '？')
          .replace(/!/g, '！')
          .replace(/;/g, '；')
          .replace(/:/g, '：')

        return formatted
      })

      // 3. 重新组合成字符串
      return formattedParagraphs.join('\n')
    },

// 提取打开视频弹窗的逻辑到单独方法
    openVideoModal() {
      // 获取当前视频进度（秒）和播放状态
      const currentTime = this.$refs.videoPlayer.currentTime || 0
      const isPlaying = this.$refs.videoPlayer.isPlaying || false

      // 显示弹窗
      this.videoModalVisible = true

      // 等待弹窗渲染完成后同步进度
      this.$nextTick(() => {
        setTimeout(() => {
          if (this.$refs.modalVideoPlayer && this.$refs.modalVideoPlayer.player) {
            const player = this.$refs.modalVideoPlayer.player
            // 检查播放器是否已准备好
            if (!player.readyState || player.readyState() < 2) {
              // 如果播放器还没准备好，等待 ready 事件
              player.ready(() => {
                try {
                  // 设置弹窗视频进度与原视频一致
                  player.currentTime(currentTime)
                  // 保持播放状态一致
                  if (isPlaying && player.paused()) {
                    const playPromise = player.play()
                    if (playPromise !== undefined) {
                      playPromise.catch(error => {
                        console.warn('播放视频失败:', error)
                      })
                    }
                  } else if (!isPlaying && !player.paused()) {
                    player.pause()
                  }
                } catch (error) {
                  console.warn('操作视频播放器失败:', error)
                }
              })
            } else {
              try {
                // 设置弹窗视频进度与原视频一致
                player.currentTime(currentTime)
                // 保持播放状态一致
                if (isPlaying && player.paused()) {
                  const playPromise = player.play()
                  if (playPromise !== undefined) {
                    playPromise.catch(error => {
                      console.warn('播放视频失败:', error)
                    })
                  }
                } else if (!isPlaying && !player.paused()) {
                  player.pause()
                }
              } catch (error) {
                console.warn('操作视频播放器失败:', error)
              }
            }
          }
        }, 100) // 短暂延迟确保DOM完全渲染
      })
    },

    // 关闭弹窗时同步进度到主视频
    handleModalClose() {
      // 1. 暂停放大的视频（核心：关闭时必停放大视频）
      if (this.$refs.modalVideoPlayer && this.$refs.modalVideoPlayer.player) {
        // 无论放大视频是否在播放，都强制暂停
        this.$refs.modalVideoPlayer.player.pause()
      }

      if (this.$refs.modalVideoPlayer && this.$refs.modalVideoPlayer.player &&
        this.$refs.videoPlayer && this.$refs.videoPlayer.player) {

        const mainPlayer = this.$refs.videoPlayer.player
        // 2. 同步放大视频的进度到原视频（确保回到原视频时进度一致）
        const modalTime = this.$refs.modalVideoPlayer.currentTime || 0
        try {
          mainPlayer.currentTime(modalTime)

          // 3. 原视频自动播放（核心：关闭放大后，原视频强制恢复播放）
          // 先判断原视频是否已暂停，避免重复触发播放导致错误
          if (mainPlayer.paused() && mainPlayer.readyState && mainPlayer.readyState() >= 2) {
            const playPromise = mainPlayer.play()
            if (playPromise !== undefined) {
              playPromise.catch(error => {
                console.warn('播放视频失败:', error)
              })
            }
          }
        } catch (error) {
          console.warn('操作视频播放器失败:', error)
        }
      }

      // 4. 关闭放大视频模态框
      this.videoModalVisible = false

      // 原有图表重新渲染逻辑（保持不变）

// 初始化关键能力二级指标时间序列图表
    initAbilitySecondaryTimeline(dom, labels, colorMap, actions) {
      let max = 0
      const lanes = labels
      const events = {}
      Object.keys(events).forEach(key => delete events[key])
      lanes.forEach(n => { events[n] = [] })

      if (!actions || !Array.isArray(actions)) {
        actions = []
      }

      // 处理数据
      actions.forEach(action => {
        const typeName = action.type
        max = action.end > max ? action.end : max
        if (lanes.includes(typeName)) {
          const startMinute = action.start
          const endMinute = action.end
          const duration = action.duration
          events[typeName].push([startMinute, lanes.indexOf(typeName), duration])
        }
      })

      const series = lanes.map(label => ({
        name: label,
        type: 'scatter',
        coordinateSystem: 'cartesian2d',
        symbol: 'roundRect',
        data: events[label].filter(item => {
          // item[1] 是 lanes.indexOf(typeName)，确保数据属于当前指标
          return lanes.indexOf(label) === item[1]
        }),
        symbolSize: (val) => [Math.max(8, val[2] * 15), 14],
        itemStyle: {
          color: colorMap[label]
        },
        emphasis: {
          disabled: true
        },
        tooltip: {
          formatter: (p) => {
            const startTime = this.formatMinuteToTime(p.value[0])
            const endTime = this.formatMinuteToTime(p.value[0] + p.value[2])
            return `${p.seriesName}：${startTime}-${endTime}`
          }
        }
      }))

      const chart = this.$echarts.init(dom)
      chart.setOption({
        grid: {
          left: 100,
          right: 20,
          top: 20,
          bottom: 30,
          height: '80%'
        },
        xAxis: {
          type: 'value',
          min: 0,
          max: Math.ceil(max),
          name: '',
          nameTextStyle: {
            color: '#9fbcd3',
            fontSize: 12
          },
          axisLine: {
            lineStyle: {
              color: '#2f4a68'
            }
          },
          axisLabel: {
            color: '#9fbcd3',
            formatter: '{value}'
          },
          splitLine: {
            lineStyle: {
              color: 'rgba(255, 255, 255, 0.06)'
            }
          },
          axisTick: {
            lineStyle: {
              color: '#9fbcd3'
            }
          }
        },
        yAxis: {
          show: true,
          type: 'category',
          data: lanes,
          inverse: true,
          axisLine: {
            lineStyle: {
              color: '#2f4a68'
            }
          },
          axisLabel: {
            color: '#9fbcd3',
            fontSize: 10,
            interval: 0
          },
          splitLine: {
            show: false
          },
          axisTick: {
            show: false
          }
        },
        tooltip: {
          trigger: 'item',
          backgroundColor: 'rgba(10, 31, 59, 0.9)',
          borderColor: '#1390AB',
          borderWidth: 1,
          textStyle: {
            color: '#fff'
          },
        },
        legend: {
          show: false,
          top: 'bottom',
          textStyle: {
            color: '#9fbcd3',
            fontSize: 10
          },
          itemWidth: 12,
          itemHeight: 8,
          type: 'scroll'
        },
        series
      })

      return chart
    },

    initWordCloudCharts (domId, data) {
      this.$nextTick(() => {
        if (!this.$echarts) return

        const dom = document.getElementById(domId)
        if (!dom) return

        const chart = this.$echarts.init(dom)
        const renderData = []

        // 处理数据：支持对象和数组两种格式
        if (Array.isArray(data)) {
          // 如果是数组格式，直接使用
          renderData.push(...data.map(item => ({
            name: typeof item === 'string' ? item : item.name || item.text,
            value: typeof item === 'string' ? 1 : item.value || item.count || 1
          })))
        } else if (typeof data === 'object' && data !== null) {
          // 如果是对象格式，转换为数组
          for (const prop in data) {
            renderData.push({
              'name': prop,
              'value': data[prop]
            })
          }
        }

        if (renderData.length === 0) {
          // 如果没有数据，显示空状态
          chart.setOption({
            graphic: {
              type: 'text',
              left: 'center',
              top: 'center',
              style: {
                text: '暂无数据',
                fill: '#9fbcd3',
                fontSize: 14,
                fontWeight: 'normal'
              }
            }
          })
          return
        }

        const option = {
          series: [
            {
              type: 'wordCloud',
              // 词云形状，可选 'circle'（圆形）、'cardioid'（心形）等，也可自定义图片形状
              shape: 'rectangle',
              // 词云布局相关配置 - 完全去除边距确保铺满
              left: 0,
              top: 0,
              right: 0,
              bottom: 0,
              width: '100%',
              height: '100%',
              // 字体大小范围 - 调整为更适合横向排列的范围

              // 字体间距 - 最小化间距

              // 关键：强制所有文字横向排列（0度）

              rotationStep: 0,
              // 允许文字超出边界以更好填充
              drawOutOfBound: false,
              // 布局算法优化
              layoutAnimation: false,

              // 词云文字样式

                // 颜色可配置渐变等，这里简单用单色
                color: function () {
                  return 'rgb(' + [
                    Math.round(Math.random() * 160),

                  // 鼠标悬浮时的放大倍数
                  shadowBlur: 10,
                  shadowColor: '#333'
                }
              }
            }
          ]
        }
        chart.setOption(option)
      })
